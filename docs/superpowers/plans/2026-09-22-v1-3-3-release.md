```ts
import { execFile } from 'node:child_process'

export interface BorrowedIdentity {
  installed: boolean
  version?: string
  userAgent?: string
  originator?: string
}

const cache = new Map<string, BorrowedIdentity>()

function run(bin: string): Promise<string | null> {
  return new Promise(resolve => {
    execFile(bin, ['--version'], { timeout: 4000, windowsHide: true }, (err, stdout) =>
      resolve(err ? null : String(stdout)))
  })
}

function parseVersion(out: string | null): string | undefined {
  if (!out) return undefined
  const m = out.match(/(\d+\.\d+\.\d+)/)
  return m ? m[1] : undefined
}

async function detect(key: string, bin: string, originator: string, opts?: { refresh?: boolean }): Promise<BorrowedIdentity> {
  if (!opts?.refresh && cache.has(key)) return cache.get(key)!
  const version = parseVersion(await run(bin))
  const identity: BorrowedIdentity = version
    ? { installed: true, version, userAgent: `${originator}/${version}`, originator }
    : { installed: false }
  cache.set(key, identity)
  return identity
}

export const detectCodexIdentity = (opts?: { refresh?: boolean }) =>
  detect('codex', 'codex', 'codex_cli', opts)
export const detectAntigravityIdentity = (opts?: { refresh?: boolean }) =>
  detect('antigravity', 'antigravity', 'antigravity', opts)
export function __resetIdentityCacheForTests(): void { cache.clear() }
```
